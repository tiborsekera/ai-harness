## .github/skills/log-extraction/execute_remote_log_search.py

```python
  1: import subprocess
  2: import shlex
  3: import json
  4: import re
  5: import sys
  6: from typing import Optional, Literal
  7: from pydantic import BaseModel, Field, ValidationError, field_validator
  8: 
  9: FORBIDDEN_SHELL_TOKENS = (
 10:     '|',
 11:     ';',
 12:     '&',
 13:     '$',
 14:     '>',
 15:     '<',
 16:     '`',
 17:     '\n',
 18:     'sudo',
 19:     'rm',
 20:     'mv',
 21:     'cp',
 22:     'tee',
 23:     'chmod',
 24:     'chown',
 25:     'wget',
 26:     'curl',
 27:     'nc',
 28:     'netcat',
 29: )
 30: 
 31: # 1. Define the strict Schema (The Contract with the LLM)
 32: class LogSearchRequest(BaseModel):
 33:     server: str = Field(..., description="Target server name")
 34:     connection_type: Literal["teleport", "direct"] = Field(..., description="Connection method")
 35:     target_path: str = Field(..., description="Full path to the remote log file")
 36:     search_pattern: str = Field(..., description="Primary pattern to grep")
 37:     filter_condition: Optional[str] = Field(None, description="Optional awk condition")
 38: 
 39:     # 2. Defense-in-Depth: Ban shell operators from the search strings
 40:     @field_validator('search_pattern', 'filter_condition')
 41:     def block_shell_metacharacters(cls, v):
 42:         if v is None:
 43:             return v
 44: 
 45:         for token in FORBIDDEN_SHELL_TOKENS:
 46:             pattern = re.escape(token)
 47:             if token.isalpha():
 48:                 pattern = rf"\b{pattern}\b"
 49: 
 50:             if re.search(pattern, v, flags=re.IGNORECASE):
 51:                 raise ValueError(f"Security Policy Violation: Forbidden shell token '{token}' found in input.")
 52:         return v
 53: 
 54:
```
